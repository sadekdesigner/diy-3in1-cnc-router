# خطوات المحاور (Steps per mm) - محسوبة لعمود T8 وموتور Nema 17 مع DRV8825 بدقة 1/32

$101=800.000    # Y-axis steps per mm
$102=800.000    # Z-axis steps per mm

# أقصى سرعات حركة سريعة (Max Feed Rate in mm/min)
$110=2500.000   # Max travel rate for X
$111=2500.000   # Max travel rate for Y
$112=800.000    # Max travel rate for Z

# التسارع (Acceleration in mm/sec^2)
$120=80.000     # Acceleration for X
$121=80.000     # Acceleration for Y
$122=50.000     # Acceleration for Z

# أقصى مسافة للمشوار الفعلي (Max Travel Distance in mm)
$130=400.000    # X max travel
$131=400.000    # Y max travel
$132=65.000     # Z max travel

# التحكم في سرعة الموتور الحافر (Spindle RPM)
$30=12000       # أقصى عدد لفات للموتور (RPM) عند كتابة S12000
$31=0           # أقل عدد لفات للموتور عند كتابة S0

# إعدادات الحساسات والحماية
$20=0           # Soft limits (0=معطل، 1=مفعل لمنع خروج الماكينة عن مساحتها)
$21=1           # Hard limits (تفعيل إيقاف الماكينة عند لمس سويتشات النهاية)
$22=1           # Homing cycle (تفعيل دورة الرجوع التلقائي لنقطة الصفر)
$23=3           # اتجاه دورة الهومينج (Homing dir invert mask)
$24=25.000      # سرعة البحث الدقيق عن سويتش البداية (Homing feed mm/min)
$25=500.000     # سرعة البحث السريع عن سويتش البداية (Homing seek mm/min)
$26=250         # تأخير الارتداد للسويتش بالمللي ثانية (Homing debounce ms)
$27=1.000       # مسافة ارتداد السويتش بعد اللمس (Homing pull-off mm)

# وضع تشغيل الماكينة
$32=0           # Laser mode enable (0 = وضع CNC راوتر عادي، 1 = وضع ليزر أو كتر استيكر)
